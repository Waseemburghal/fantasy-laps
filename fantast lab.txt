<script>
"use strict";
// ================= CONFIG =================
const SHEET_ID = "1Vph49aLPd_jaezMYwCi2WrlbNhaGv1MAsTCupBkcUlQ"; // must be shared: Anyone with link -> Viewer
const REFRESH_MS = 60000;
const BOTTOM_COUNT = 3; // teams highlighted red at the bottom
const TOP_COUNT = 3;    // teams highlighted gold at the top
// ==========================================

const state = { tab: "table", gw: null, data: { table: [], matches: [], managers: { gws: [], rows: [] } }, error: null };
const view = document.getElementById("view");

async function gviz(sheet) {
  const url = `https://docs.google.com/spreadsheets/d/${SHEET_ID}/gviz/tq?tqx=out:json&sheet=${encodeURIComponent(sheet)}`;
  const res = await fetch(url, { cache: "no-store", credentials: "omit" });
  if (!res.ok) throw new Error(sheet + ": " + res.status);
  const txt = await res.text();
  const json = JSON.parse(txt.slice(txt.indexOf("{"), txt.lastIndexOf("}") + 1));
  if (json.status === "error") throw new Error(sheet + ": " + (json.errors && json.errors[0] && json.errors[0].message));
  return json.table;
}
const val = (c) => (c && c.v !== null && c.v !== undefined ? c.v : null);
const num = (v) => { const n = Number(v); return Number.isFinite(n) ? n : null; };

async function loadTable() {
  const t = await gviz("League_table");
  // Script writes: C=rank, D=team, E=P, F=W, G=D, H=L, I=PTS, J=FPL
  return t.rows.map(r => r.c || []).map(c => ({
    pos: num(val(c[2])), team: val(c[3]), p: num(val(c[4])), w: num(val(c[5])),
    d: num(val(c[6])), l: num(val(c[7])), pts: num(val(c[8])), fpl: num(val(c[9]))
  })).filter(r => r.pos !== null && r.team);
}

async function loadMatches() {
  const t = await gviz("Matches");
  // A=GW, B=home, C=homeScore, D=awayScore, E=away
  return t.rows.map(r => r.c || []).map(c => ({
    gw: num(val(c[0])), home: val(c[1]), hs: num(val(c[2])), as: num(val(c[3])), away: val(c[4])
  })).filter(m => m.gw !== null && m.home && m.away);
}

async function loadManagers() {
  const t = await gviz("FPL_Managers");
  const gwCols = [];
  t.cols.forEach((col, i) => { const m = /^GW\s*(\d+)$/i.exec(col.label || ""); if (m) gwCols.push({ i, gw: +m[1] }); });
  const rows = t.rows.map(r => r.c || []).map(c => {
    const pts = gwCols.map(g => num(val(c[g.i])));
    return { team: val(c[0]), name: val(c[1]), pts, total: pts.reduce((s, x) => s + (x || 0), 0) };
  }).filter(r => r.team && r.name);
  return { gws: gwCols.map(g => g.gw), rows };
}

// ---------- DOM helpers (textContent only => no HTML injection) ----------
function el(tag, props = {}, ...kids) {
  const e = document.createElement(tag);
  for (const [k, v] of Object.entries(props)) {
    if (k === "class") e.className = v; else if (k === "text") e.textContent = v; else e.setAttribute(k, v);
  }
  kids.forEach(k => k && e.append(k));
  return e;
}
const td = (text, cls) => el("td", { class: cls || "", text: text === null || text === undefined ? "–" : String(text) });

function renderTable() {
  const rows = state.data.table;
  if (!rows.length) return el("div", { class: "empty", text: "ما في بيانات بعد." });
  const head = el("tr", {}, ...[["#"], ["الفريق", "team"], ["لعب"], ["ف"], ["ت"], ["خ"], ["نقاط"], ["FPL"]]
    .map(([t, c]) => el("th", { class: c || "", text: t })));
  const body = el("tbody");
  rows.forEach((r, i) => {
    const cls = i < TOP_COUNT ? "top" : (i >= rows.length - BOTTOM_COUNT ? "bottom" : "");
    body.append(el("tr", { class: cls },
      td(r.pos, "pos"), td(r.team, "team"), td(r.p), td(r.w), td(r.d), td(r.l), td(r.pts, "pts"), td(r.fpl)));
  });
  return el("div", { class: "wrap" }, el("table", {}, el("thead", {}, head), body));
}

function renderMatches() {
  const all = state.data.matches;
  if (!all.length) return el("div", { class: "empty", text: "ما في نتائج بعد." });
  const gws = [...new Set(all.map(m => m.gw))].sort((a, b) => a - b);
  if (state.gw === null || !gws.includes(state.gw)) state.gw = gws[gws.length - 1];
  const chips = el("div", { class: "chips" }, ...gws.map(g => {
    const b = el("button", { type: "button", "aria-pressed": String(g === state.gw), text: "GW " + g });
    b.addEventListener("click", () => { state.gw = g; render(); });
    return b;
  }));
  const list = el("div");
  all.filter(m => m.gw === state.gw).forEach(m => {
    const hw = m.hs !== null && m.as !== null && m.hs > m.as;
    const aw = m.hs !== null && m.as !== null && m.as > m.hs;
    list.append(el("div", { class: "match" },
      el("div", { class: "t home" + (hw ? " win" : ""), text: m.home }),
      el("div", { class: "score", text: `${m.hs ?? "-"} : ${m.as ?? "-"}` }),
      el("div", { class: "t away" + (aw ? " win" : ""), text: m.away })));
  });
  return el("div", {}, chips, list);
}

function renderManagers() {
  const { gws, rows } = state.data.managers;
  if (!rows.length) return el("div", { class: "empty", text: "ما في بيانات بعد." });
  const sorted = [...rows].sort((a, b) => b.total - a.total);
  const head = el("tr", {}, el("th", { text: "#" }), el("th", { class: "team", text: "المدير" }),
    el("th", { class: "team", text: "الفريق" }), ...gws.map(g => el("th", { text: "GW" + g })), el("th", { text: "المجموع" }));
  const body = el("tbody");
  sorted.forEach((r, i) => {
    body.append(el("tr", {}, td(i + 1, "pos"), td(r.name, "team"), td(r.team, "team"),
      ...r.pts.map(p => td(p)), td(r.total, "pts")));
  });
  return el("div", { class: "wrap" }, el("table", {}, el("thead", {}, head), body));
}

function render() {
  document.querySelectorAll("nav button").forEach(b => b.setAttribute("aria-selected", String(b.dataset.tab === state.tab)));
  view.replaceChildren();
  if (state.error) { view.append(el("div", { class: "err", text: "تعذّر تحميل البيانات. تأكد إن الشيت مشارك (Anyone with the link → Viewer)." })); return; }
  view.append(state.tab === "table" ? renderTable() : state.tab === "matches" ? renderMatches() : renderManagers());
}

async function refresh() {
  const label = document.getElementById("updated");
  try {
    const [table, matches, managers] = await Promise.all([loadTable(), loadMatches(), loadManagers()]);
    state.data = { table, matches, managers };
    state.error = null;
    label.textContent = "آخر تحديث: " + new Date().toLocaleTimeString("ar-JO", { hour: "2-digit", minute: "2-digit" });
  } catch (e) {
    console.error(e);
    state.error = e;
    label.textContent = "فشل التحديث";
  }
  render();
}

document.querySelectorAll("nav button").forEach(b => b.addEventListener("click", () => { state.tab = b.dataset.tab; render(); }));
document.getElementById("refresh").addEventListener("click", refresh);
document.addEventListener("visibilitychange", () => { if (!document.hidden) refresh(); });
setInterval(() => { if (!document.hidden) refresh(); }, REFRESH_MS);
refresh();
</script>
